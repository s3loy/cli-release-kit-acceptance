cmake_minimum_required(VERSION 3.20)
project(bird VERSION 1.2.3 LANGUAGES C)
if(NOT BIRD_ACCEPTANCE_MESSAGE STREQUAL "source-preset-checked")
  message(FATAL_ERROR "The tracked source preset was not applied")
endif()
add_executable(native-cli main.c)
set_target_properties(native-cli PROPERTIES OUTPUT_NAME bird-on-disk)
if(MSVC)
  set_property(TARGET native-cli PROPERTY MSVC_RUNTIME_LIBRARY "${BIRD_CRT}")
endif()
target_compile_definitions(native-cli PRIVATE APP_VERSION="${PROJECT_VERSION}" APP_NAME="${BIRD_APP_NAME}" PRESET_MESSAGE="${BIRD_ACCEPTANCE_MESSAGE}")
